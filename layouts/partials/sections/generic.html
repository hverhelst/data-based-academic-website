{{/* The fallback renderer: turns any array of flat records into the same resume-item markup the built-in presets use, driven entirely by the field names declared in the section's registry entry. This is what makes a brand-new section possible without writing a template. Drop data/grants.json next to an entry in data/sections.json and the section renders; write layouts/partials/sections/grants.html later only if the generic shape is not good enough. Every field is looked up by name and skipped when absent, so a record that carries only a title still renders cleanly. Expects: .section (a registry entry), .pages (the records). */}} {{ $s := .section }} {{ $records := .pages | default slice }} {{ with $s.sortField }} {{ $records = sort $records . (cond $s.sortReverse "desc" "asc") }} {{ end }} {{/* With no groupField every record lands in one unnamed group, so the same loop below handles grouped and ungrouped sections alike. */}} {{ $groups := slice (dict "name" "" "records" $records) }} {{ with $s.groupField }} {{ $field := . }} {{ $names := slice }} {{ range $records }} {{ $v := index . $field }} {{ if and $v (not (in $names $v)) }}{{ $names = $names | append $v }}{{ end }} {{ end }} {{ $grouped := slice }} {{ range $names }} {{ $grouped = $grouped | append (dict "name" . "records" (where $records $field .)) }} {{ end }} {{/* Records missing the grouping field would silently vanish otherwise. */}} {{ $ungrouped := where $records $field nil }} {{ with $ungrouped }} {{ $grouped = $grouped | append (dict "name" "" "records" .) }} {{ end }} {{ $groups = $grouped }} {{ end }}
